$ curl -X PUT localhost:8789/stock/widget -d '{"qty":10}' \
    -H "Authorization: Bearer $ROOT"
200 10

$ curl -X PUT localhost:8789/orders/o1 \
    -H "Authorization: Bearer $ALICE" \
    -d '{"sku":"widget","qty":3,"cents":2500}'
201 {"id":"o1","status":"placed","owner":"alice","sku":"widget",
    "qty":3,"cents":2500}

$ curl localhost:8788/totals -H "Authorization: Bearer $ROOT"
200 {"orders":1,"units":3,"cents":7500}

$ curl -X POST localhost:8789/orders/o1/cancel \
    -H "Authorization: Bearer $ALICE"
200 {"id":"o1","status":"cancelled","owner":"alice","sku":"widget",
    "qty":3,"cents":2500}

$ curl localhost:8788/totals -H "Authorization: Bearer $ROOT"
200 {"orders":0,"units":0,"cents":0}

$ curl localhost:8788/totals -H "Authorization: Bearer $ALICE"
403
